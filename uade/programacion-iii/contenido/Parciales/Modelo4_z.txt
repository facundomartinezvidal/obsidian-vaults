Algoritmo(P1, P2, P3, P4, int P)
	crearMatriz(M)
	for i in range(n) //cantidad de objetos
		for j in range(P)
			if (j == 0)
				M[i,j] = 0
			if (i == 0)
				if (Pi.Precio <= j)
					M[i,j] = Pi.Precio * Pi.Descuento
				else
					M[i,j] = 0
			else
				if (Pi.Precio <= j)
					M[i,j] = max(M[i-1,j-Pi.Precio] + Pi.Precio*Pi.Descuento, M[i-1,j])
				else
					M[i,j] = M[i-1,j]
	return M[n,P]

Algoritmo(int[] A, int q) {
        for(int i = 0; i < 2; i++){
            if (q == 0){
                break;
            }else if (i == 2-1) {
                for (int j = 0; j < q; j++)
                    A[i] = A[i] * -1;
            }else
                A[i] = A[i] * -1;
