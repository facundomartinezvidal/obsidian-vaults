EJERCICIO 2:
presupuesto P = 10
(p)  3 9 5 3 gramos
(c)  4 1 2 4 calorias por gramo
(v)  6 7 9 7 precio

Algoritmo(int[][] G)
	crearmatriz(M)
	for i in range (G[0].length)
		for j in range (P+1)
			if (j == 0)
				M[i,j] = 0
			else if (v[i] <= j)
				if (i == 0)
					M[i,j] = M[ i, j-v[i] ] + p[i]*c[i]
				else
					M[i,j] = max( M[ i, j-v[i] ] + p[i]*c[i], M[i-1, j] )
			else
				if (i == 0)
					M[i,j] = 0
				else
					M[i,j] = M[i-1, j]
O(n*P)

es greedy
Peso maximo por contenedor: P = 10			
 9  8  7  6  5  4  3  O objetos (peso de cada uno) ordenado de mayor a menor
-1 -1 -1 -1 -1 -1 -1  S (numero del contenedor)
 0  0  0  0  0  0  0  C contenedores (peso acumulado)
 
Algoritmo(int[] O, P)
	crear arreglo S con n posiciones lleno de -1
	crear arreglo C con n posiciones lleno de 0 // conjunto de candidatos
	for i in range(n)
		for j in range(i)             // funcion seleccion siempre el objeto mas pesado
			if (C[j] + O[i] <= P && j < min)  // funcion factibilidad
				min = j //guardamos el primer contenedor anterior que entre el objeto
		S[i] = j
		C[j] += O[i]

O(n2)
